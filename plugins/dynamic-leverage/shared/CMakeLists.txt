cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dynamic_leverage_core VERSION 0.2.0 LANGUAGES CXX)
  include(CTest)
endif()

option(DYNAMIC_LEVERAGE_BUILD_RULES_CHECK "Build offline native rules validation tool" ON)
if(DYNAMIC_LEVERAGE_BUILD_RULES_CHECK)
  add_executable(DynamicLeverageRulesCheck tools/rules_check.cpp)
  target_link_libraries(DynamicLeverageRulesCheck PRIVATE dynamic_leverage_core)
  set_target_properties(DynamicLeverageRulesCheck PROPERTIES CXX_EXTENSIONS OFF
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_compile_options(DynamicLeverageRulesCheck PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(DynamicLeverageRulesCheck PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

add_library(dynamic_leverage_core STATIC src/core.cpp src/rules_json.cpp src/opening_policy.cpp)
target_include_directories(dynamic_leverage_core PUBLIC include)
target_compile_features(dynamic_leverage_core PUBLIC cxx_std_17)
set_target_properties(dynamic_leverage_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(dynamic_leverage_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(dynamic_leverage_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
  add_executable(dynamic_leverage_core_tests tests/core_tests.cpp)
  target_link_libraries(dynamic_leverage_core_tests PRIVATE dynamic_leverage_core)
  set_target_properties(dynamic_leverage_core_tests PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME dynamic_leverage_core_tests COMMAND dynamic_leverage_core_tests)
  add_executable(dynamic_leverage_rules_tests tests/rules_tests.cpp)
  target_link_libraries(dynamic_leverage_rules_tests PRIVATE dynamic_leverage_core)
  set_target_properties(dynamic_leverage_rules_tests PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_compile_options(dynamic_leverage_rules_tests PRIVATE /utf-8)
  endif()
  add_test(NAME dynamic_leverage_rules_tests COMMAND dynamic_leverage_rules_tests)
  add_executable(dynamic_leverage_opening_policy_tests tests/opening_policy_tests.cpp)
  target_link_libraries(dynamic_leverage_opening_policy_tests PRIVATE dynamic_leverage_core)
  set_target_properties(dynamic_leverage_opening_policy_tests PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME dynamic_leverage_opening_policy_tests COMMAND dynamic_leverage_opening_policy_tests)
endif()
