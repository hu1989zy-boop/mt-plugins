cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dynamic_leverage_mt5 VERSION 0.2.0 LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET dynamic_leverage_core)
  add_subdirectory(../shared "${CMAKE_CURRENT_BINARY_DIR}/shared")
endif()
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  foreach(_test IN ITEMS adapter events discovery order_margin)
    add_executable(dynamic_leverage_mt5_${_test}_tests tests/${_test}_tests.cpp)
    target_include_directories(dynamic_leverage_mt5_${_test}_tests PRIVATE src)
    target_link_libraries(dynamic_leverage_mt5_${_test}_tests PRIVATE dynamic_leverage_core Threads::Threads)
    set_target_properties(dynamic_leverage_mt5_${_test}_tests PROPERTIES
      CXX_EXTENSIONS OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
      target_compile_options(dynamic_leverage_mt5_${_test}_tests PRIVATE /W4 /permissive- /utf-8)
    else()
      target_compile_options(dynamic_leverage_mt5_${_test}_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME dynamic_leverage_mt5_${_test}_tests COMMAND dynamic_leverage_mt5_${_test}_tests)
  endforeach()
  add_executable(dynamic_leverage_mt5_probe_settings_tests probe/settings_tests.cpp)
  target_compile_features(dynamic_leverage_mt5_probe_settings_tests PRIVATE cxx_std_17)
  set_target_properties(dynamic_leverage_mt5_probe_settings_tests PROPERTIES
    CXX_EXTENSIONS OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME dynamic_leverage_mt5_probe_settings_tests COMMAND dynamic_leverage_mt5_probe_settings_tests)
endif()

option(DYNAMIC_LEVERAGE_MT5_BUILD_SERVER "Build MT5 Server API DLL using Windows MSVC x64" OFF)
option(DYNAMIC_LEVERAGE_MT5_BUILD_MARGIN_PROBE "Build read-only MT5 order margin diagnostic DLL (Windows MSVC x64)" OFF)
if(DYNAMIC_LEVERAGE_MT5_BUILD_SERVER OR DYNAMIC_LEVERAGE_MT5_BUILD_MARGIN_PROBE)
  set(MT5_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/mt5-sdk" CACHE PATH "Licensed MT5 SDK root")
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MT5 DLL requires Windows MSVC x64; configure mt5/ with -A x64")
  endif()
  if(NOT EXISTS "${MT5_SDK_ROOT}/Include/MT5APIServer.h")
    message(FATAL_ERROR "MT5_SDK_ROOT must contain Include/MT5APIServer.h")
  endif()
  file(STRINGS "${MT5_SDK_ROOT}/Include/MT5APIServer.h" _version REGEX "^#define[ \t]+MTServerAPIVersion[ \t]+[0-9]+")
  if(NOT _version MATCHES "MTServerAPIVersion[ \t]+6182([ \t]|$)")
    message(FATAL_ERROR "This adapter targets the unmodified MT5 Server API 6182 headers")
  endif()
endif()
if(DYNAMIC_LEVERAGE_MT5_BUILD_SERVER)
  add_library(DynamicLeverageMT5 SHARED src/plugin.cpp)
  target_include_directories(DynamicLeverageMT5 SYSTEM PRIVATE "${MT5_SDK_ROOT}/Include")
  target_link_libraries(DynamicLeverageMT5 PRIVATE dynamic_leverage_core)
  target_compile_features(DynamicLeverageMT5 PRIVATE cxx_std_17)
  target_compile_definitions(DynamicLeverageMT5 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_compile_options(DynamicLeverageMT5 PRIVATE /W4 /permissive- /utf-8)
  set_target_properties(DynamicLeverageMT5 PROPERTIES PREFIX "" CXX_EXTENSIONS OFF
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  include(../shared/cmake/plugin_validation.cmake)
  dynamic_leverage_validate_dll(DynamicLeverageMT5 mt5 MTServerAbout MTServerCreate)
endif()

if(DYNAMIC_LEVERAGE_MT5_BUILD_MARGIN_PROBE)
  add_library(DynamicLeverageMT5MarginProbe SHARED probe/plugin.cpp)
  target_include_directories(DynamicLeverageMT5MarginProbe SYSTEM PRIVATE "${MT5_SDK_ROOT}/Include")
  target_include_directories(DynamicLeverageMT5MarginProbe PRIVATE src)
  target_link_libraries(DynamicLeverageMT5MarginProbe PRIVATE dynamic_leverage_core)
  target_compile_features(DynamicLeverageMT5MarginProbe PRIVATE cxx_std_17)
  target_compile_definitions(DynamicLeverageMT5MarginProbe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_compile_options(DynamicLeverageMT5MarginProbe PRIVATE /W4 /permissive- /utf-8)
  set_target_properties(DynamicLeverageMT5MarginProbe PROPERTIES PREFIX "" CXX_EXTENSIONS OFF
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  include(../shared/cmake/plugin_validation.cmake)
  dynamic_leverage_validate_dll(DynamicLeverageMT5MarginProbe mt5_margin_probe MTServerAbout MTServerCreate)
endif()
