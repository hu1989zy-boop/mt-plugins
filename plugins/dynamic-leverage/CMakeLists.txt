cmake_minimum_required(VERSION 3.21)
project(dynamic_leverage VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(shared)
option(DYNAMIC_LEVERAGE_BUILD_MT4 "Build MT4 adapter and portable tests" ON)
option(DYNAMIC_LEVERAGE_BUILD_MT5 "Build MT5 adapter and portable tests" ON)
if(DYNAMIC_LEVERAGE_BUILD_MT4)
  add_subdirectory(mt4)
endif()
if(DYNAMIC_LEVERAGE_BUILD_MT5)
  add_subdirectory(mt5)
endif()

option(DYNAMIC_LEVERAGE_TEST_MANAGEMENT_CONTRACT "Test saved management JSON against native rule engine when Python is available" ON)
if(BUILD_TESTING AND DYNAMIC_LEVERAGE_TEST_MANAGEMENT_CONTRACT AND TARGET DynamicLeverageRulesCheck)
  find_package(Python3 3.10 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME dynamic_leverage_management_contract
      COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/management/tests/native_contract.py"
              --checker "$<TARGET_FILE:DynamicLeverageRulesCheck>")
    set_tests_properties(dynamic_leverage_management_contract PROPERTIES TIMEOUT 60)
  else()
    message(STATUS "Python 3.10+ not found: management/native file contract test not registered")
  endif()
endif()
