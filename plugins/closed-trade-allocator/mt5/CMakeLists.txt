cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(closed_trade_allocator_mt5 VERSION 0.1.0 LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET closedalloc_core)
  add_subdirectory(../shared "${CMAKE_CURRENT_BINARY_DIR}/shared")
endif()
if(BUILD_TESTING)
  add_executable(closedalloc_mt5_model_tests tests/model_tests.cpp)
  target_include_directories(closedalloc_mt5_model_tests PRIVATE src)
  target_link_libraries(closedalloc_mt5_model_tests PRIVATE closedalloc_core)
  add_test(NAME closedalloc_mt5_model_tests COMMAND closedalloc_mt5_model_tests)
  find_package(Threads REQUIRED)
  add_executable(closedalloc_mt5_events_tests tests/events_tests.cpp)
  target_include_directories(closedalloc_mt5_events_tests PRIVATE src)
  target_compile_features(closedalloc_mt5_events_tests PRIVATE cxx_std_17)
  target_link_libraries(closedalloc_mt5_events_tests PRIVATE Threads::Threads)
  add_test(NAME closedalloc_mt5_events_tests COMMAND closedalloc_mt5_events_tests)
  add_executable(closedalloc_mt5_source_work_tests tests/source_work_tests.cpp)
  target_include_directories(closedalloc_mt5_source_work_tests PRIVATE src)
  target_compile_features(closedalloc_mt5_source_work_tests PRIVATE cxx_std_17)
  add_test(NAME closedalloc_mt5_source_work_tests COMMAND closedalloc_mt5_source_work_tests)
endif()
option(CLOSEDALLOC_MT5_BUILD_SERVER "Build the MT5 Server API DLL (MSVC x64)" OFF)
if(CLOSEDALLOC_MT5_BUILD_SERVER)
  set(MT5_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/mt5-sdk" CACHE PATH "Licensed MT5 SDK root containing Include/MT5APIServer.h")
  if(NOT EXISTS "${MT5_SDK_ROOT}/Include/MT5APIServer.h")
    message(FATAL_ERROR "Set MT5_SDK_ROOT to your licensed MT5 Server API SDK")
  endif()
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MT5 Server API plugin requires Windows MSVC x64; configure mt5/ with -A x64")
  endif()
  file(STRINGS "${MT5_SDK_ROOT}/Include/MT5APIServer.h" _version REGEX "^#define[ \t]+MTServerAPIVersion[ \t]+[0-9]+")
  if(NOT _version MATCHES "MTServerAPIVersion[ \t]+6182([ \t]|$)")
    message(FATAL_ERROR "This adapter requires unmodified MTServerAPIVersion 6182 headers")
  endif()
  add_library(ClosedTradeAllocatorMT5 SHARED src/plugin.cpp)
  target_include_directories(ClosedTradeAllocatorMT5 SYSTEM PRIVATE "${MT5_SDK_ROOT}/Include")
  target_link_libraries(ClosedTradeAllocatorMT5 PRIVATE closedalloc_core bcrypt)
  target_compile_features(ClosedTradeAllocatorMT5 PRIVATE cxx_std_17)
  target_compile_definitions(ClosedTradeAllocatorMT5 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601)
  set_target_properties(ClosedTradeAllocatorMT5 PROPERTIES PREFIX "" CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(ClosedTradeAllocatorMT5 PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_compile_options(ClosedTradeAllocatorMT5 PRIVATE -Wall -Wextra)
  endif()
else()
  message(STATUS "MT5 DLL disabled; enable CLOSEDALLOC_MT5_BUILD_SERVER with Windows MSVC x64")
endif()

# Enforce one CRT for the entire link, including targets created by shared/ or
# an enclosing project. Setting only the DLL leaves older core builds on /MD.
foreach(_closedalloc_target IN ITEMS closedalloc_core closedalloc_core_tests
    closedalloc_mt5_model_tests closedalloc_mt5_events_tests closedalloc_mt5_source_work_tests ClosedTradeAllocatorMT5)
  if(TARGET ${_closedalloc_target})
    set_property(TARGET ${_closedalloc_target} PROPERTY
      MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
      get_target_property(_closedalloc_source_dir ${_closedalloc_target} SOURCE_DIR)
      message(STATUS "${_closedalloc_target}: static CRT (/MT, Debug /MTd); source=${_closedalloc_source_dir}")
    endif()
  endif()
endforeach()
