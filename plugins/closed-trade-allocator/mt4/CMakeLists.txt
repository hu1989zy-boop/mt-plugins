cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(mt4_closed_trade_allocator VERSION 0.1.0 LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET closedalloc_core)
  add_subdirectory(../shared "${CMAKE_CURRENT_BINARY_DIR}/shared")
endif()
add_library(mt4_closedalloc_model STATIC model.cpp)
target_include_directories(mt4_closedalloc_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mt4_closedalloc_model PUBLIC closedalloc_core)
target_compile_features(mt4_closedalloc_model PUBLIC cxx_std_17)
set_target_properties(mt4_closedalloc_model PROPERTIES CXX_EXTENSIONS OFF
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(mt4_closedalloc_model PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mt4_closedalloc_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(mt4_closedalloc_tests tests/model_tests.cpp)
  target_link_libraries(mt4_closedalloc_tests PRIVATE mt4_closedalloc_model)
  set_target_properties(mt4_closedalloc_tests PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME mt4_closedalloc_tests COMMAND mt4_closedalloc_tests)
endif()
option(CLOSEDALLOC_MT4_BUILD_SERVER "Build the MT4 Server API DLL (MSVC Win32)" OFF)
set(MT4_SERVER_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/mt4-sdk" CACHE PATH "Proprietary MT4 Server API SDK")
if(CLOSEDALLOC_MT4_BUILD_SERVER)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "MT4ClosedTradeAllocator requires Windows MSVC Win32; configure mt4/ with -A Win32")
  endif()
  set(_header "${MT4_SERVER_SDK_DIR}/include/MT4ServerAPI.h")
  if(NOT EXISTS "${_header}")
    message(FATAL_ERROR "Missing proprietary SDK: ${_header}")
  endif()
  file(STRINGS "${_header}" _version REGEX "^#define[ \t]+ServerApiVersion[ \t]+[0-9]+")
  if(NOT _version MATCHES "ServerApiVersion[ \t]+5([ \t]|$)")
    message(FATAL_ERROR "This adapter requires unmodified ServerApiVersion 5 headers")
  endif()
  add_library(MT4ClosedTradeAllocator SHARED plugin.cpp sdk_port.cpp plugin.def)
  target_include_directories(MT4ClosedTradeAllocator SYSTEM PRIVATE "${MT4_SERVER_SDK_DIR}/include")
  target_link_libraries(MT4ClosedTradeAllocator PRIVATE mt4_closedalloc_model bcrypt)
  target_compile_definitions(MT4ClosedTradeAllocator PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _USE_32BIT_TIME_T _WIN32_WINNT=0x0601 WINVER=0x0601)
  target_compile_options(MT4ClosedTradeAllocator PRIVATE /W4 /permissive- /utf-8 /EHa)
  set_target_properties(MT4ClosedTradeAllocator PROPERTIES PREFIX "" CXX_EXTENSIONS OFF
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
