cmake_minimum_required(VERSION 3.21)
project(mt4_trade_copier VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_compile_definitions(_USE_32BIT_TIME_T)
endif()
include(CTest)
option(MT4_TRADECOPIER_BUILD_WINDOWS_UI "Build the MT4 Windows manager" ${WIN32})
option(MT4_TRADECOPIER_BUILD_EXECUTOR "Build the MT4 Server API 5 DLL (default disarmed)" OFF)
get_filename_component(_mt4_repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MT4_SERVER_SDK_DIR "${_mt4_repo}/vendor/mt4-sdk" CACHE PATH "MT4 SDK containing include/MT4ServerAPI.h")

set(_mt4_shared_cmake "${CMAKE_CURRENT_SOURCE_DIR}/../shared/cmake/Core.cmake")
if(NOT EXISTS "${_mt4_shared_cmake}")
  message(FATAL_ERROR
    "Incomplete MT4 source tree: missing ${_mt4_shared_cmake}\n"
    "Copy the complete shared/ directory alongside mt4/ from the same source revision.\n"
    "mt4/ and shared/ must be siblings; copying mt4/ alone is not sufficient.")
endif()
include("${_mt4_shared_cmake}")
tradecopier_add_core(TARGET_PREFIX mt4_tradecopier PLATFORM mt4
  EXAMPLE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/copier.example.json")

add_library(mt4_tradecopier_port STATIC core/src/port.cpp)
target_include_directories(mt4_tradecopier_port PUBLIC core/include)
target_link_libraries(mt4_tradecopier_port PUBLIC mt4_tradecopier_execution)
target_compile_features(mt4_tradecopier_port PUBLIC cxx_std_17)
set_target_properties(mt4_tradecopier_port PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(mt4_tradecopier_port PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mt4_tradecopier_port PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(MT4TradeCopierJournal tools/inspect_journal.cpp)
target_link_libraries(MT4TradeCopierJournal PRIVATE mt4_tradecopier_execution)
if(MINGW)
  target_link_options(MT4TradeCopierJournal PRIVATE -municode)
endif()

if(WIN32)
  add_library(mt4_tradecopier_management_pipe STATIC management/local_pipe.cpp)
  target_include_directories(mt4_tradecopier_management_pipe PUBLIC management)
  target_link_libraries(mt4_tradecopier_management_pipe PUBLIC mt4_tradecopier_execution advapi32)
  target_compile_definitions(mt4_tradecopier_management_pipe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 WINVER=0x0601)
  if(MSVC)
    target_compile_options(mt4_tradecopier_management_pipe PRIVATE /W4 /permissive- /utf-8)
  endif()
endif()
if(MT4_TRADECOPIER_BUILD_WINDOWS_UI)
  add_subdirectory(windows)
endif()
if(MT4_TRADECOPIER_BUILD_EXECUTOR)
  add_subdirectory(server)
endif()

if(BUILD_TESTING)
  add_executable(mt4_tradecopier_management_ui_tests tests/management_ui_tests.cpp)
  target_link_libraries(mt4_tradecopier_management_ui_tests PRIVATE mt4_tradecopier_execution)
  target_include_directories(mt4_tradecopier_management_ui_tests PRIVATE windows)
  if(MSVC)
    target_compile_options(mt4_tradecopier_management_ui_tests PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(mt4_tradecopier_management_ui_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME mt4_tradecopier_management_ui_tests COMMAND mt4_tradecopier_management_ui_tests)

  add_executable(mt4_tradecopier_port_tests tests/port_tests.cpp)
  target_link_libraries(mt4_tradecopier_port_tests PRIVATE mt4_tradecopier_port)
  target_include_directories(mt4_tradecopier_port_tests PRIVATE server)
  add_test(NAME mt4_tradecopier_port_tests COMMAND mt4_tradecopier_port_tests)
  add_executable(mt4_tradecopier_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(mt4_tradecopier_runtime_tests PRIVATE mt4_tradecopier_execution)
  target_include_directories(mt4_tradecopier_runtime_tests PRIVATE server)
  add_test(NAME mt4_tradecopier_runtime_tests COMMAND mt4_tradecopier_runtime_tests)
  foreach(dealer_test dealer_policy dealer_execution)
    add_executable(mt4_tradecopier_${dealer_test}_tests tests/${dealer_test}_tests.cpp)
    target_compile_features(mt4_tradecopier_${dealer_test}_tests PRIVATE cxx_std_17)
    target_include_directories(mt4_tradecopier_${dealer_test}_tests PRIVATE server)
    if(MSVC)
      target_compile_options(mt4_tradecopier_${dealer_test}_tests PRIVATE /W4 /permissive- /utf-8)
    else()
      target_compile_options(mt4_tradecopier_${dealer_test}_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME mt4_tradecopier_${dealer_test}_tests COMMAND mt4_tradecopier_${dealer_test}_tests)
  endforeach()
endif()

if(WIN32 AND BUILD_TESTING)
  add_executable(mt4_tradecopier_management_pipe_tests tests/management_pipe_tests.cpp)
  target_link_libraries(mt4_tradecopier_management_pipe_tests PRIVATE mt4_tradecopier_management_pipe)
  target_compile_definitions(mt4_tradecopier_management_pipe_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  if(MSVC)
    target_compile_options(mt4_tradecopier_management_pipe_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_test(NAME mt4_tradecopier_management_pipe_tests COMMAND mt4_tradecopier_management_pipe_tests)
  set_tests_properties(mt4_tradecopier_management_pipe_tests PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
