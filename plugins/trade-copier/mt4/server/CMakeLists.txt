if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "MT4TradeCopierExecutor requires Windows MSVC Win32; use a separate -A Win32 build from mt4/.")
endif()
set(_header "${MT4_SERVER_SDK_DIR}/include/MT4ServerAPI.h")
if(NOT EXISTS "${_header}")
  message(FATAL_ERROR "MT4 SDK header not found: ${_header}")
endif()
file(STRINGS "${_header}" _version REGEX "^#define[ \t]+ServerApiVersion[ \t]+[0-9]+")
if(NOT _version MATCHES "ServerApiVersion[ \t]+5([ \t]|$)")
  message(FATAL_ERROR "This MT4 adapter targets ServerApiVersion 5. Do not change the SDK version macro.")
endif()
add_library(MT4TradeCopierExecutor SHARED executor.cpp sdk_broker.cpp executor.def)
target_include_directories(MT4TradeCopierExecutor SYSTEM PRIVATE "${MT4_SERVER_SDK_DIR}/include")
target_link_libraries(MT4TradeCopierExecutor PRIVATE mt4_tradecopier_port mt4_tradecopier_management_pipe bcrypt)
target_compile_features(MT4TradeCopierExecutor PRIVATE cxx_std_17)
target_compile_definitions(MT4TradeCopierExecutor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _USE_32BIT_TIME_T _WIN32_WINNT=0x0601 WINVER=0x0601)
target_compile_options(MT4TradeCopierExecutor PRIVATE /W4 /permissive- /utf-8 /EHa)
set_target_properties(MT4TradeCopierExecutor PROPERTIES PREFIX "" CXX_EXTENSIONS OFF)
