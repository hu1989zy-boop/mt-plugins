if(NOT WIN32)
  message(FATAL_ERROR "MT4TradeCopierManager requires Windows and the Win32 SDK")
endif()

add_executable(MT4TradeCopierManager WIN32 main.cpp)
target_link_libraries(MT4TradeCopierManager PRIVATE mt4_tradecopier_management_pipe comctl32 comdlg32 user32 gdi32 uxtheme)
target_compile_features(MT4TradeCopierManager PRIVATE cxx_std_17)
target_compile_definitions(MT4TradeCopierManager PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601)
set_target_properties(MT4TradeCopierManager PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(MT4TradeCopierManager PRIVATE /W4 /permissive- /utf-8)
elseif(MINGW)
  target_compile_options(MT4TradeCopierManager PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(MT4TradeCopierManager PRIVATE -municode)
endif()
