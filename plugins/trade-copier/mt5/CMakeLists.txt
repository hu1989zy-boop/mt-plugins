cmake_minimum_required(VERSION 3.21)
project(mt5_trade_copier VERSION 0.7.4 LANGUAGES CXX)

# Match the supplied Server API sample's /MT (/MTd for Debug) runtime.
# Keep this consistent across the UI and the static libraries it consumes.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRADECOPIER_BUILD_WINDOWS_UI "Build the native Windows manager" ${WIN32})
option(TRADECOPIER_BUILD_SDK_PROBE "Build the read-only Server API loading/event probe" OFF)
option(TRADECOPIER_BUILD_EXECUTOR "Build the Server API 6182 trading executor (default disarmed)" OFF)
get_filename_component(_tradecopier_repository_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MT5_SERVER_SDK_DIR "${_tradecopier_repository_root}/vendor/mt5-sdk" CACHE PATH
    "Supplied MT5 SDK root containing Include/MT5APIServer.h")

include(CTest)
include("${CMAKE_CURRENT_SOURCE_DIR}/../shared/cmake/Core.cmake")
tradecopier_add_core(TARGET_PREFIX tradecopier PLATFORM mt5
  EXAMPLE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/copier.example.json")

if(TRADECOPIER_BUILD_WINDOWS_UI)
  add_subdirectory(windows)
endif()
add_executable(TradeCopierJournal tools/inspect_journal.cpp)
target_link_libraries(TradeCopierJournal PRIVATE tradecopier_execution)
if(MINGW)
  target_link_options(TradeCopierJournal PRIVATE -municode)
endif()
if(BUILD_TESTING)
  add_executable(tradecopier_stress_simulation tests/stress_simulation.cpp)
  target_link_libraries(tradecopier_stress_simulation PRIVATE tradecopier_execution)
  add_executable(tradecopier_execution_tests tests/execution_tests.cpp)
  target_link_libraries(tradecopier_execution_tests PRIVATE tradecopier_execution)
  target_include_directories(tradecopier_execution_tests PRIVATE server)
  if(MSVC)
    target_compile_options(tradecopier_execution_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_test(NAME tradecopier_execution_tests COMMAND tradecopier_execution_tests)
  foreach(readonly_test readonly_policy readonly_refusal history_index journal_performance journal_codec performance_log)
    add_executable(tradecopier_${readonly_test}_tests tests/${readonly_test}_tests.cpp)
    target_link_libraries(tradecopier_${readonly_test}_tests PRIVATE tradecopier_execution)
    target_include_directories(tradecopier_${readonly_test}_tests PRIVATE server)
    if(MSVC)
      target_compile_options(tradecopier_${readonly_test}_tests PRIVATE /W4 /permissive- /utf-8)
    else()
      target_compile_options(tradecopier_${readonly_test}_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME tradecopier_${readonly_test}_tests COMMAND tradecopier_${readonly_test}_tests)
  endforeach()
  add_executable(tradecopier_direction_journal_tests tests/direction_journal_tests.cpp)
  target_link_libraries(tradecopier_direction_journal_tests PRIVATE tradecopier_execution)
  add_test(NAME tradecopier_direction_journal_tests COMMAND tradecopier_direction_journal_tests)
  add_executable(tradecopier_management_ui_tests tests/management_ui_tests.cpp)
  target_link_libraries(tradecopier_management_ui_tests PRIVATE tradecopier_execution)
  target_include_directories(tradecopier_management_ui_tests PRIVATE windows)
  if(MSVC)
    target_compile_options(tradecopier_management_ui_tests PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(tradecopier_management_ui_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME tradecopier_management_ui_tests COMMAND tradecopier_management_ui_tests)
endif()
if(TRADECOPIER_BUILD_SDK_PROBE OR TRADECOPIER_BUILD_EXECUTOR)
  add_subdirectory(server)
endif()

if(WIN32)
  add_library(tradecopier_management_pipe STATIC management/local_pipe.cpp)
  target_include_directories(tradecopier_management_pipe PUBLIC management)
  target_link_libraries(tradecopier_management_pipe PUBLIC tradecopier_execution advapi32)
  target_compile_definitions(tradecopier_management_pipe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 WINVER=0x0601)
  if(MSVC)
    target_compile_options(tradecopier_management_pipe PRIVATE /W4 /permissive- /utf-8)
  endif()
endif()

if(WIN32 AND BUILD_TESTING)
  add_executable(tradecopier_management_pipe_tests tests/management_pipe_tests.cpp)
  target_link_libraries(tradecopier_management_pipe_tests PRIVATE tradecopier_management_pipe)
  target_compile_definitions(tradecopier_management_pipe_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  add_test(NAME tradecopier_management_pipe_tests COMMAND tradecopier_management_pipe_tests)
  set_tests_properties(tradecopier_management_pipe_tests PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
